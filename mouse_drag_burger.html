// 1. 오디오 컨텍스트 생성 (크로스 브라우징) const AudioContext = window.AudioContext || window.webkitAudioContext; let audioCtx = null; // 오디오 엔진 초기화 및 잠금 해제 함수 function initAudio() { if (!audioCtx) { audioCtx = new AudioContext(); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } } // 화면 첫 터치 / 클릭 시 모바일 오디오 잠금 해제 window.addEventListener('touchstart', initAudio, { once: true }); window.addEventListener('click', initAudio, { once: true }); // 2. 톤 사운드 재생 함수 (재생 직전 항상 활성화 확인) function playTone(freq, type, duration) { try { initAudio(); if (!audioCtx) return; const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); osc.type = type; osc.frequency.setValueAtTime(freq, audioCtx.currentTime); gain.gain.setValueAtTime(0.15, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration); osc.connect(gain); gain.connect(audioCtx.destination); osc.start(); osc.stop(audioCtx.currentTime + duration); } catch(e) { console.warn("오디오 재생 실패:", e); } }
